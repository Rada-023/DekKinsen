        function updateModalQty(delta) {
            modalQty = Math.max(1, modalQty + delta);
            document.getElementById('modal-qty').innerText = modalQty;
        }

        // Shopping Cart Logic
        function quickAddToCart(id) {
            addToCart(id, 1);
        }

        function addToCart(id, qty) {
            const item = menuItems.find(m => m.id === id);
            const existing = cart.find(c => c.id === id);

            if (existing) {
                existing.qty += qty;
            } else {
                cart.push({ ...item, qty });
            }

            updateCartUI();
        }

        function updateCartUI() {
            const badge = document.getElementById('cart-badge');
            const totalQty = cart.reduce((sum, item) => sum + item.qty, 0);

            if (totalQty > 0) {
                badge.innerText = totalQty;
                badge.classList.remove('hidden');
            } else {
                badge.classList.add('hidden');
            }

            // Render Cart Drawer Items
            const container = document.getElementById('cart-items');
            container.innerHTML = '';

            if (cart.length === 0) {
                container.innerHTML = `<div class="text-center text-gray-500 text-sm py-12">ไม่มีรายการในตะกร้า</div>`;
            } else {
                cart.forEach(item => {
                    const row = document.createElement('div');
                    row.className = "flex items-center gap-3 p-3 rounded-xl bg-white/5 border border-white/5";
                    row.innerHTML = `
                        <img src="${item.image}" class="w-14 h-14 rounded-lg object-cover">
                        <div class="flex-1">
                            <h5 class="text-xs font-bold text-white line-clamp-1">${item.title}</h5>
                            <span class="text-xs text-brand-gold">${item.price} THB x ${item.qty}</span>
                        </div>
                        <div class="flex items-center gap-1 bg-black/40 rounded-lg p-1">
                            <button onclick="changeCartQty(${item.id}, -1)" class="px-1.5 py-0.5 text-xs text-gray-300 hover:text-white">-</button>
                            <span class="text-xs text-white px-1 font-semibold">${item.qty}</span>
                            <button onclick="changeCartQty(${item.id}, 1)" class="px-1.5 py-0.5 text-xs text-gray-300 hover:text-white">+</button>
                        </div>
                    `;
                    container.appendChild(row);
                });
            }

            // Calculate Total
            const total = cart.reduce((sum, item) => sum + (item.price * item.qty), 0);
            document.getElementById('cart-total').innerText = `${total} THB`;
        }

        function changeCartQty(id, delta) {
            const item = cart.find(c => c.id === id);
            if (!item) return;

            item.qty += delta;
            if (item.qty <= 0) {
                cart = cart.filter(c => c.id !== id);
            }
            updateCartUI();
        }

        function toggleCartDrawer() {
            const drawer = document.getElementById('cart-drawer');
            drawer.classList.toggle('hidden');
        }

        function checkout() {
            if (cart.length === 0) {
                alert('โปรดเลือกรายการอาหารก่อนทำการสั่งซื้อ');
                return;
            }
            alert('ขอบคุณสำหรับคำสั่งซื้อ! DekKinsen AI กำลังเตรียมเมนูสุดพิเศษให้คุณ');
            cart = [];
            updateCartUI();
            toggleCartDrawer();
        }

        // AI Noodle Recommendation Assistant Algorithm
        function runAIAssistant() {
            const mood = document.getElementById('ai-mood').value;
            const health = document.getElementById('ai-health').value;

            let recommendedId = 1;

            if (mood === 'spicy' && health === 'detox') recommendedId = 1;
            else if (mood === 'healthy' || health === 'lowcarbs') recommendedId = 4;
            else if (mood === 'exotic' || health === 'antioxidants') recommendedId = 3;
            else if (health === 'protein') recommendedId = 5;
            else if (mood === 'healthy' || mood === 'exotic') recommendedId = 2;

            const recItem = menuItems.find(item => item.id === recommendedId);

            const resultBox = document.getElementById('ai-result');
            document.getElementById('ai-rec-img').style.backgroundImage = `url('${recItem.image}')`;
            document.getElementById('ai-rec-title').innerText = recItem.title;
            document.getElementById('ai-rec-desc').innerText = `${recItem.health} (ราคา ${recItem.price} THB)`;

            const recBtn = document.getElementById('ai-rec-btn');
            recBtn.onclick = () => {
                addToCart(recItem.id, 1);
                toggleCartDrawer();
            };

            resultBox.classList.remove('hidden');
        }
